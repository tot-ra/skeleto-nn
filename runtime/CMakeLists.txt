cmake_minimum_required(VERSION 3.25)
project(skeleto_runtime LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED MOTIONBRICKS_SOURCE)
    message(FATAL_ERROR "Set MOTIONBRICKS_SOURCE to the pinned motion-bricks.cpp checkout")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MOTIONBRICKS_DOWNLOAD_MODELS OFF CACHE BOOL "" FORCE)
set(MOTIONBRICKS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MOTIONBRICKS_ENABLE_VULKAN OFF CACHE BOOL "" FORCE)
set(MOTIONBRICKS_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${MOTIONBRICKS_SOURCE}" motionbricks EXCLUDE_FROM_ALL)
add_executable(skeleto_motion_worker native_worker.cpp)
target_link_libraries(skeleto_motion_worker PRIVATE motionbricks_static)
target_include_directories(skeleto_motion_worker PRIVATE "${MOTIONBRICKS_SOURCE}/include")
if(APPLE)
    set_target_properties(skeleto_motion_worker PROPERTIES INSTALL_RPATH "@loader_path")
endif()
